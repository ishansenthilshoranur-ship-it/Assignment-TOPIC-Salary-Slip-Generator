#include <stdio.h> #define MAX_EMPLOYEES 50 int main(void) {    char names[MAX_EMPLOYEES][50];    float basicPay[MAX_EMPLOYEES];    int count = 0, choice = 0, searchSno = 0, i = 0;    do {        printf("\n=== SALARY SLIP GENERATOR ===\n");        printf("1. Add Employee\n2. View All Employees\n3. Search Employee by S.No.\n4. Exit\n");        printf("Enter choice (1-4): ");        scanf("%d", &choice);        switch (choice) {            case 1: /* Add employee: name, basic pay */                if (count >= MAX_EMPLOYEES) {                    printf("Employee list is full! Maximum %d records allowed.\n", MAX_EMPLOYEES);                } else {                    printf("Enter employee name: ");                    scanf(" %49[^\n]", names[count]);                    printf("Enter basic pay: ");                    scanf("%f", &basicPay[count]);                    count++;                    printf("Employee added! Assigned S.No.: %d\n", count);                }                break;            case 2: /* View all employees */                if (count == 0) {                    printf("No employee records found.\n");                } else {                    printf("\n%-8s %-28s %12s\n", "S.No.", "Employee Name", "Basic Pay");                    printf("--------------------------------------------------\n");                    for (i = 0; i < count; i++) {                        printf("%-8d %-28s %12.2f\n", i + 1, names[i], basicPay[i]);                    }                }                break;            case 3: /* Search employee by S.No. */                if (count == 0) {                    printf("No employee records available to search.\n");                } else {                    printf("Enter S.No. to search: ");                    scanf("%d", &searchSno);                    if (searchSno < 1 || searchSno > count) {                        printf("Employee with S.No. %d does not exist.\n", searchSno);                    } else {                        printf("S.No.: %d | Name: %s | Basic Pay: %.2f\n",                               searchSno, names[searchSno - 1], basicPay[searchSno - 1]);                    }
